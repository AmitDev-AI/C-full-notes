4. Grade calculator Easy

Convert a score from 0–100 to a letter grade: A ≥ 90, B ≥ 80, C ≥ 70, D ≥ 60, otherwise F. Reject scores outside 0–100.

Related: C If...Else

Show solution 
#include <stdio.h>

int main(void) {
    int score;

    printf("Score (0-100): ");
    scanf("%d", &score);

    if (score < 0 || score > 100) {
        printf("Invalid score.\n");
    } else if (score >= 90) {
        printf("Grade A\n");
    } else if (score >= 80) {
        printf("Grade B\n");
    } else if (score >= 70) {
        printf("Grade C\n");
    } else if (score >= 60) {
        printf("Grade D\n");
    } else {
        printf("Grade F\n");
    }

    return 0;
}

7. Largest of three Easy

    printf("Three integers: ");
    scanf("%d %d %d", &a, &b, &c);

    max = a;
    if (b > max) max = b;
    if (c > max) max = c;
    printf("if version: %d\n", max);

    max = (a > b) ? a : b;
    max = (c > max) ? c : max;
    printf("ternary version: %d\n", max);

}

8. Simple calculator Easy

Read two numbers and an operator (+ - * /) and print the result using a switch. Handle division by zero.

Related: C Switch

Show solution 
#include <stdio.h>

int main(void) {
    double a, b;
    char op;

    printf("Expression (for example 12.5 * 4): ");
    scanf("%lf %c %lf", &a, &op, &b);

    switch (op) {
        case '+': printf("%.2f\n", a + b); break;
        case '-': printf("%.2f\n", a - b); break;
        case '*': printf("%.2f\n", a * b); break;
        case '/':
            if (b == 0) printf("Cannot divide by zero.\n");
            else printf("%.2f\n", a / b);
            break;
        default: printf("Unknown operator.\n");
    }

    return 0;
}

9. Array statistics Medium

Read 10 integers into an array and print the sum, average (as a double), minimum and maximum.

Related: C Arrays

Show solution 
#include <stdio.h>

int main(void) {
    int values[10];
    int i, sum = 0, min, max;

    printf("Enter 10 integers:\n");
    for (i = 0; i < 10; i++) {
        scanf("%d", &values[i]);
    }

    min = max = values[0];
    for (i = 0; i < 10; i++) {
        sum += values[i];
        if (values[i] < min) min = values[i];
        if (values[i] > max) max = values[i];
    }

    printf("sum     = %d\n", sum);
    printf("average = %.2f\n", sum / 10.0);
    printf("min     = %d\n", min);
    printf("max     = %d\n", max);

    return 0;
}

11. Count vowels Medium

Count the vowels (a, e, i, o, u) in a sentence read with fgets. Treat uppercase letters as vowels too.

Related: C Strings · C User Input

Show solution 
#include <stdio.h>
#include <ctype.h>

int main(void) {
    char text[200];
    int i, count = 0;

    printf("Sentence: ");
    fgets(text, sizeof(text), stdin);

    for (i = 0; text[i] != '\0'; i++) {
        char c = (char)tolower((unsigned char)text[i]);
        if (c == 'a' || c == 'e' || c == 'i' || c == 'o' || c == 'u')
            count++;
    }

    printf("Vowels: %d\n", count);
    return 0;
}

12. Palindrome check Medium

Read a word and decide whether it reads the same forwards and backwards. Ignore case.

Related: C Strings

Show solution 
#include <stdio.h>
#include <string.h>
#include <ctype.h>

int main(void) {
    char word[100];
    int left, right, ok = 1;

    printf("Word: ");
    scanf("%99s", word);

    left = 0;
    right = (int)strlen(word) - 1;
    while (left < right) {
        if (tolower((unsigned char)word[left]) !=
            tolower((unsigned char)word[right])) {
            ok = 0;
            break;
        }
        left++;
        right--;
    }

    printf("%s is %sa palindrome.\n", word, ok ? "" : "not ");
    return 0;
}

13. Factorial and Fibonacci Medium

Write two functions: one returns n! with a loop, the other returns the nth Fibonacci number with recursion. Print the first 10 values of each.

Related: C Functions

Show solution 
#include <stdio.h>

long factorial(int n) {
    long result = 1;
    int i;
    for (i = 2; i <= n; i++) result *= i;
    return result;
}

long fibonacci(int n) {
    if (n < 2) return n;
    return fibonacci(n - 1) + fibonacci(n - 2);
}

int main(void) {
    int i;
    for (i = 0; i < 10; i++) {
        printf("%d! = %ld   fib(%d) = %ld\n",
               i, factorial(i), i, fibonacci(i));
    }
    return 0;
}

14. Swap with pointers Medium

void swap(int *a, int *b) {
    int temp = *a;
    *a = *b;
    *b = temp;
}

int main(void) {
    int x = 10, y = 99;
    printf("before: x=%d y=%d\n", x, y);
    swap(&x, &y);
    printf("after:  x=%d y=%d\n", x, y);

15. Bubble sort Medium

Sort an array of 8 integers in ascending order using bubble sort, then print the result.

Related: C Arrays · C Loops

Show solution 
#include <stdio.h>

int main(void) {
    int a[8] = {5, 2, 9, 1, 7, 3, 8, 4};
    int i, j, temp;

    for (i = 0; i < 7; i++) {
        for (j = 0; j < 7 - i; j++) {
            if (a[j] > a[j + 1]) {
                temp = a[j];
                a[j] = a[j + 1];
                a[j + 1] = temp;
            }
        }
    }

    for (i = 0; i < 8; i++) printf("%d ", a[i]);
    printf("\n");
    return 0;
}

17. Dynamic array Hard

Ask for a count, allocate an int array with malloc, fill it, print it, then free it. Check that allocation succeeded.

Related: C Memory

Show solution 
#include <stdio.h>
#include <stdlib.h>

int main(void) {
    int n, i;
    int *values;

    printf("How many numbers? ");
    scanf("%d", &n);
    if (n <= 0) {
        printf("Nothing to do.\n");
        return 1;
    }

    values = malloc((size_t)n * sizeof(int));
    if (values == NULL) {
        printf("Out of memory.\n");
        return 1;
    }

    for (i = 0; i < n; i++) {
        printf("value %d: ", i + 1);
        scanf("%d", &values[i]);
    }

    printf("You entered: ");
    for (i = 0; i < n; i++) printf("%d ", values[i]);
    printf("\n");

    free(values);
    return 0;
}

18. Prime sieve Hard

Print every prime below 100 using the Sieve of Eratosthenes. Use an array of flags and a loop that starts at i * i.

Related: C Arrays · C Loops

Show solution 
#include <stdio.h>

int main(void) {
    int is_prime[100];
    int i, j;

    for (i = 0; i < 100; i++) is_prime[i] = 1;
    is_prime[0] = is_prime[1] = 0;

    for (i = 2; i * i < 100; i++) {
        if (is_prime[i]) {
            for (j = i * i; j < 100; j += i) {
                is_prime[j] = 0;
            }
        }
    }

    for (i = 2; i < 100; i++) {
        if (is_prime[i]) printf("%d ", i);
    }
    printf("\n");
    return 0;
}
