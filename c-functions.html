C Functions

A function is a named block of code that performs a task. Functions let you reuse logic, keep main short, and test pieces of a program independently.

Defining and calling a function

#include <stdio.h>

// return_type name(parameters) { body }
void greet(void) {
    printf("Hello from a function!\n");
}

int main(void) {
    greet();      // call it
    greet();      // call it again
    return 0;
}
Output
Hello from a function!
Hello from a function!
  • void as the return type means the function returns no value.
  • (void) means it takes no parameters.
  • A function must be declared before it is called — either define it above the call or write a prototype.

Parameters and return values

#include <stdio.h>

int add(int a, int b) {
    return a + b;
}

double average(double a, double b) {
    return (a + b) / 2.0;
}

void print_line(int length) {
    for (int i = 0; i < length; i++) {
        putchar('-');
    }
    putchar('\n');
}

int main(void) {
    int sum = add(12, 30);
    printf("sum = %d\n", sum);
    printf("average = %.2f\n", average(10, 15));
    print_line(20);
    return 0;
}
Output
sum = 42
average = 12.50
--------------------

Function prototypes

A prototype tells the compiler the function's signature before its body appears. It lets you put main first and helper functions below.

#include <stdio.h>

// prototypes
int square(int n);
void show_result(int value);

int main(void) {
    show_result(square(7));
    return 0;
}

int square(int n) {
    return n * n;
}

void show_result(int value) {
    printf("Result: %d\n", value);
}
Output
Result: 49

Pass by value

C passes arguments by value: the function receives copies. Changing a parameter does not change the caller's variable.

#include <stdio.h>

void try_change(int x) {
    x = 100;
    printf("inside: %d\n", x);
}

int main(void) {
    int n = 5;
    try_change(n);
    printf("outside: %d\n", n);   // still 5
    return 0;
}
Output
inside: 100
outside: 5

To let a function modify the caller's variable, pass its address and accept a pointer — see the Pointers chapter.

void set_to_100(int *x) {
    *x = 100;
}

int main(void) {
    int n = 5;
    set_to_100(&n);
    printf("%d\n", n);    // 100
    return 0;
}

Returning values

int max(int a, int b) {
    return a > b ? a : b;
}

// A function can return early
int safe_divide(int a, int b) {
    if (b == 0) {
        return 0;         // caller should check
    }
    return a / b;
}

// void returns nothing
void log_message(void) {
    puts("log entry");
    // return; is optional
}

Return at most one value in C. To return several results, use pointers as output parameters or return a structure.

Local scope

Variables declared inside a function are local and disappear when it returns. Each call gets a fresh copy.

int counter(void) {
    int count = 0;    // new every call
    count++;
    return count;
}

// counter() always returns 1.

Use static to keep a local variable alive between calls:

#include <stdio.h>

int counter(void) {
    static int count = 0;   // initialised once
    count++;
    return count;
}

int main(void) {
    printf("%d\n", counter());   // 1
    printf("%d\n", counter());   // 2
    printf("%d\n", counter());   // 3
    return 0;
}
Output
1
2
3

Recursion

A recursive function calls itself. Every recursive function needs a base case that stops the recursion.

#include <stdio.h>

long long factorial(int n) {
    if (n <= 1) {
        return 1;              // base case
    }
    return n * factorial(n - 1);  // recursive case
}

int main(void) {
    printf("5! = %lld\n", factorial(5));
    printf("10! = %lld\n", factorial(10));
    return 0;
}
Output
5! = 120
10! = 3628800
// Fibonacci — simple but slow (exponential)
int fib(int n) {
    if (n <= 1) return n;
    return fib(n - 1) + fib(n - 2);
}

// Iterative version — fast
int fib_fast(int n) {
    int a = 0, b = 1;
    for (int i = 0; i < n; i++) {
        int next = a + b;
        a = b;
        b = next;
    }
    return a;
}

Warning: each recursive call uses stack memory. Deep recursion can overflow the stack and crash. Prefer iteration when the depth may be large.

Functions and arrays

#include <stdio.h>

int sum_array(const int values[], int length) {
    int total = 0;
    for (int i = 0; i < length; i++) {
        total += values[i];
    }
    return total;
}

int main(void) {
    int data[] = {1, 2, 3, 4, 5};
    int n = sizeof(data) / sizeof(data[0]);
    printf("sum = %d\n", sum_array(data, n));
    return 0;
}
Output
sum = 15

Multiple source files

Large programs are split across files. Put declarations in a header and definitions in a .c file.

#ifndef MATH_UTILS_H
#define MATH_UTILS_H

int add(int a, int b);
int multiply(int a, int b);

#endif
#include "math_utils.h"

int add(int a, int b) {
    return a + b;
}

int multiply(int a, int b) {
    return a * b;
}
#include <stdio.h>
#include "math_utils.h"

int main(void) {
    printf("%d\n", add(3, 4));
    printf("%d\n", multiply(3, 4));
    return 0;
}

Compile all the source files together: gcc main.c math_utils.c -o app.

Good function design

  • One job per function. If you need "and" to describe it, split it.
  • Short parameter lists. More than four or five parameters suggests a structure.
  • Clear names. calculate_average beats calc.
  • Document behaviour — especially edge cases and error returns.
  • Avoid hidden state. Prefer parameters and return values over globals.
