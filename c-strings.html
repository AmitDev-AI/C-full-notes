C Strings

C has no separate string type. A string is an array of char ending with a special null character '\0'. Everything about strings follows from that one fact.

String literals and character arrays

#include <stdio.h>

int main(void) {
    char greeting[] = "Hello";     // array of 6 chars: H e l l o \0
    char name[20] = "Amit";        // room for 19 chars + terminator
    const char *ptr = "World";     // pointer to a read-only literal

    printf("%s\n", greeting);
    printf("%s\n", name);
    printf("%s\n", ptr);
    printf("size of greeting = %zu\n", sizeof(greeting));
    return 0;
}
Output
Hello
Amit
World
size of greeting = 6

The null terminator

The array has five visible letters but six bytes. Functions such as printf keep reading until they find '\0'. Without it, they run past the end of the array.

char word[] = {'C', 'a', 't', '\0'};

printf("%s\n", word);          // Cat
printf("size = %zu\n", sizeof(word));   // 4
printf("length = %zu\n", strlen(word)); // 3
Output
Cat
size = 4
length = 3

Copying strings

#include <stdio.h>
#include <string.h>

int main(void) {
    char source[] = "C programming";
    char copy[50];

    strcpy(copy, source);
    printf("copy: %s\n", copy);

    char safe[10];
    strncpy(safe, source, sizeof(safe) - 1);
    safe[sizeof(safe) - 1] = '\0';    // guarantee termination
    printf("safe: %s\n", safe);
    return 0;
}
Output
copy: C programming
safe: C program

You cannot assign strings with = after declaration. Use strcpy, or assign a new pointer.

char a[20];
// a = "hello";        // error

strcpy(a, "hello");    // correct

char b[20] = "hello";  // correct at declaration

const char *p = "hello";  // p points at a literal; do not modify it

Concatenating strings

#include <stdio.h>
#include <string.h>

int main(void) {
    char result[50] = "Hello, ";
    strcat(result, "World");
    strcat(result, "!");
    printf("%s\n", result);
    return 0;
}
Output
Hello, World!

The destination must have enough space for the original text, the appended text and the terminator. Buffer overflows from unchecked strcat/strcpy are a classic security bug.

Comparing strings

== compares addresses, not contents. Use strcmp.

#include <stdio.h>
#include <string.h>

int main(void) {
    char a[] = "apple";
    char b[] = "apple";
    char c[] = "banana";

    printf("a == b gives %d\n", a == b);        // 0: different arrays
    printf("strcmp(a,b) = %d\n", strcmp(a, b)); // 0: equal contents
    printf("strcmp(a,c) = %d\n", strcmp(a, c)); // negative

    if (strcmp(a, b) == 0) {
        printf("a and b contain the same text\n");
    }
    return 0;
}
Output
a == b gives 0
strcmp(a,b) = 0
strcmp(a,c) = -1
a and b contain the same text

Reading strings

#include <stdio.h>

int main(void) {
    char name[50];

    printf("Enter your name: ");
    if (fgets(name, sizeof(name), stdin) != NULL) {
        // fgets keeps the newline; remove it
        size_t len = strlen(name);
        if (len > 0 && name[len - 1] == '\n') {
            name[len - 1] = '\0';
        }
        printf("Hello, %s!\n", name);
    }
    return 0;
}

Prefer fgets over scanf("%s", ...): it accepts spaces and respects the buffer size. See User Input.

Arrays of strings

#include <stdio.h>

int main(void) {
    const char *days[] = {
        "Monday", "Tuesday", "Wednesday",
        "Thursday", "Friday", "Saturday", "Sunday"
    };

    for (int i = 0; i < 7; i++) {
        printf("%d: %s\n", i + 1, days[i]);
    }
    return 0;
}
Output
1: Monday
2: Tuesday
3: Wednesday
4: Thursday
5: Friday
6: Saturday
7: Sunday

Useful character functions (<ctype.h>)

FunctionTrue when
isalpha(c)Letter
isdigit(c)Digit 0–9
isalnum(c)Letter or digit
isspace(c)Space, tab, newline, etc.
isupper(c) / islower(c)Upper / lower case
toupper(c) / tolower(c)Convert case
 
#include <stdio.h>
#include <ctype.h>

int main(void) {
    char text[] = "Hello, World 123";
    int letters = 0, digits = 0;

    for (int i = 0; text[i] != '\0'; i++) {
        if (isalpha((unsigned char)text[i])) letters++;
        if (isdigit((unsigned char)text[i])) digits++;
    }

    printf("letters = %d, digits = %d\n", letters, digits);
    return 0;
}
Output
letters = 10, digits = 3

Tip: cast to unsigned char when calling <ctype.h> functions. Passing a negative char is undefined behaviour.
