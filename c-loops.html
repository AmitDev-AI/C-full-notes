C Loops

Loops repeat a block of code. C has three loop statements — while, do...while and for — plus break and continue to control them.

while loop

Checks the condition before each iteration. If it is false at the start, the body never runs.

#include <stdio.h>

int main(void) {
    int i = 1;

    while (i <= 5) {
        printf("%d\n", i);
        i++;
    }

    return 0;
}
Output
1
2
3
4
5

Forgetting i++ makes the loop run forever.

do...while loop

Checks the condition after the body, so the body runs at least once. Note the semicolon after while (...).

#include <stdio.h>

int main(void) {
    int i = 10;

    do {
        printf("i = %d\n", i);
        i++;
    } while (i < 5);

    return 0;
}
Output
i = 10

for loop

The most common loop. It packs initialisation, condition and update into one line.

for (initialise; condition; update) {
    body;
}
#include <stdio.h>

int main(void) {
    for (int i = 1; i <= 5; i++) {
        printf("%d squared = %d\n", i, i * i);
    }
    return 0;
}
Output
1 squared = 1
2 squared = 4
3 squared = 9
4 squared = 16
5 squared = 25

Variations

// Count down
for (int i = 10; i > 0; i--) {
    printf("%d ", i);
}

// Step by 2
for (int i = 0; i <= 10; i += 2) {
    printf("%d ", i);
}

// Multiple variables
for (int i = 0, j = 10; i < j; i++, j--) {
    printf("%d-%d ", i, j);
}

// Infinite (use break to exit)
for (;;) {
    break;
}

Which loop should I use?

LoopUse when
forThe number of iterations is known, or there is a clear counter.
whileRepeat while a condition holds; may run zero times.
do...whileThe body must run at least once (menus, input validation).

Nested loops

A loop inside a loop. The inner loop completes fully for each iteration of the outer loop.

#include <stdio.h>

int main(void) {
    for (int row = 1; row <= 3; row++) {
        for (int col = 1; col <= 4; col++) {
            printf("%d-%d ", row, col);
        }
        printf("\n");
    }
    return 0;
}
Output
1-1 1-2 1-3 1-4
2-1 2-2 2-3 2-4
3-1 3-2 3-3 3-4
// A triangle of stars
for (int i = 1; i <= 5; i++) {
    for (int j = 1; j <= i; j++) {
        printf("*");
    }
    printf("\n");
}
Output
*
**
***
****
*****

break

Leaves the innermost loop immediately. Execution continues after the loop.

#include <stdio.h>

int main(void) {
    for (int i = 1; i <= 10; i++) {
        if (i == 5) {
            break;
        }
        printf("%d\n", i);
    }
    printf("Done\n");
    return 0;
}
Output
1
2
3
4
Done

continue

Skips the rest of the current iteration and goes to the next one.

#include <stdio.h>

int main(void) {
    for (int i = 1; i <= 10; i++) {
        if (i % 2 == 0) {
            continue;      // skip even numbers
        }
        printf("%d\n", i);
    }
    return 0;
}
Output
1
3
5
7
9

Common loop patterns

Sum and average

#include <stdio.h>

int main(void) {
    int sum = 0;
    int n = 10;

    for (int i = 1; i <= n; i++) {
        sum += i;
    }

    printf("sum = %d\n", sum);
    printf("average = %.2f\n", (double)sum / n);
    return 0;
}
Output
sum = 55
average = 5.50

Factorial

#include <stdio.h>

int main(void) {
    int n = 5;
    unsigned long long fact = 1;

    for (int i = 1; i <= n; i++) {
        fact *= i;
    }

    printf("%d! = %llu\n", n, fact);
    return 0;
}
Output
5! = 120

Reverse a number

#include <stdio.h>

int main(void) {
    int n = 1234;
    int reversed = 0;

    while (n > 0) {
        int digit = n % 10;
        reversed = reversed * 10 + digit;
        n /= 10;
    }

    printf("reversed = %d\n", reversed);
    return 0;
}
Output
reversed = 4321

Input validation loop

int number;

do {
    printf("Enter a positive number: ");
    if (scanf("%d", &number) != 1) {
        while (getchar() != '\n') { }  // discard bad input
        number = -1;
    }
} while (number <= 0);

Common mistakes

  • Off-by-one — <= vs <. Choose deliberately.
  • Forgetting to update the counter — infinite loop.
  • Semicolon after the loop header — for (i = 0; i < 5; i++); creates an empty loop.
  • Modifying the loop variable inside the body — can cause skipped or repeated iterations.
  • Using the loop variable after the loop — in C99 its scope ends with the loop.
  • break only exits one loop — nested loops need a flag or goto.

Warning: an infinite loop is a logic bug, but some programs intentionally loop forever until break or user input. Make the exit condition obvious.
