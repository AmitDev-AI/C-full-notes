C Structures

A structure (struct) groups variables of different types under one name. It is how C models real-world objects: a point, a student, a bank account.

Defining a structure

struct Point {
    int x;
    int y;
};

This defines a new type, struct Point, with two members. It does not create any variable yet. Note the semicolon after the closing brace.

Creating and using a structure

#include <stdio.h>

struct Point {
    int x;
    int y;
};

int main(void) {
    struct Point p;          // declare
    p.x = 10;                // assign members
    p.y = 20;

    printf("(%d, %d)\n", p.x, p.y);

    struct Point q = {3, 7}; // initialise
    printf("(%d, %d)\n", q.x, q.y);
    return 0;
}
Output
(10, 20)
(3, 7)

Access members with the dot operator ..

Initialising structures

struct Point {
    int x;
    int y;
};

struct Point a = {1, 2};          // positional
struct Point b = {.y = 2, .x = 1};// designated (C99)
struct Point c = {0};             // all members zero

// Assignment copies all members
struct Point d = a;
d.x = 99;                         // a is unchanged
#include <stdio.h>

int main(void) {
    struct Point a = {1, 2};
    struct Point b = a;      // copy

    b.x = 99;
    printf("a.x = %d, b.x = %d\n", a.x, b.x);
    return 0;
}
Output
a.x = 1, b.x = 99

A realistic example

#include <stdio.h>
#include <string.h>

struct Student {
    char name[50];
    int age;
    double gpa;
};

int main(void) {
    struct Student s;

    strcpy(s.name, "Amit");
    s.age = 20;
    s.gpa = 8.75;

    printf("Name: %s\n", s.name);
    printf("Age:  %d\n", s.age);
    printf("GPA:  %.2f\n", s.gpa);
    return 0;
}
Output
Name: Amit
Age:  20
GPA:  8.75

Pointers to structures

Use -> instead of (*ptr).:

#include <stdio.h>

struct Point { int x, y; };

int main(void) {
    struct Point p = {4, 9};
    struct Point *ptr = &p;

    printf("%d\n", ptr->x);      // 4
    printf("%d\n", (*ptr).y);    // 9 — same thing

    ptr->x = 40;
    printf("%d\n", p.x);         // 40
    return 0;
}
Output
4
9
40

Arrays of structures

#include <stdio.h>

struct Product {
    char name[30];
    double price;
};

int main(void) {
    struct Product items[3] = {
        {"Pen", 1.50},
        {"Notebook", 4.25},
        {"Bag", 25.00}
    };

    double total = 0;
    for (int i = 0; i < 3; i++) {
        printf("%-10s $%.2f\n", items[i].name, items[i].price);
        total += items[i].price;
    }

    printf("Total: $%.2f\n", total);
    return 0;
}
Output
Pen        $1.50
Notebook   $4.25
Bag        $25.00
Total: $30.75

Nested structures

struct Date {
    int day, month, year;
};

struct Employee {
    char name[40];
    struct Date hired;
    double salary;
};

struct Employee e = {
    "Rina",
    {15, 6, 2021},
    52000.0
};

printf("%s hired %02d/%02d/%d\n",
       e.name, e.hired.day, e.hired.month, e.hired.year);
Output
Rina hired 15/06/2021

Structures and functions

Passing a structure by value copies all its members. For large structures, pass a pointer instead — preferably const if the function only reads.

#include <stdio.h>

struct Point { int x, y; };

struct Point make_point(int x, int y) {
    struct Point p = {x, y};
    return p;
}

void move(struct Point *p, int dx, int dy) {
    p->x += dx;
    p->y += dy;
}

int distance_sq(const struct Point *a, const struct Point *b) {
    int dx = a->x - b->x;
    int dy = a->y - b->y;
    return dx * dx + dy * dy;
}

int main(void) {
    struct Point a = make_point(1, 2);
    struct Point b = {5, 6};

    move(&a, 2, 3);
    printf("a = (%d, %d)\n", a.x, a.y);
    printf("distance squared = %d\n", distance_sq(&a, &b));
    return 0;
}
Output
a = (3, 5)
distance squared = 5

typedef: shorter names

typedef creates an alias, so you can write Point instead of struct Point.

typedef struct {
    int x;
    int y;
} Point;

Point p = {1, 2};
printf("%d\n", p.x);

You can also define the struct and the alias together, as above. The structure tag is then optional.

Self-referential structures

A structure can contain a pointer to another structure of the same type — the basis of linked lists and trees.

struct Node {
    int value;
    struct Node *next;   // pointer to the same type
};

int main(void) {
    struct Node a = {1, NULL};
    struct Node b = {2, NULL};
    a.next = &b;

    printf("%d -> %d\n", a.value, a.next->value);
    return 0;
}
Output
1 -> 2

Size, padding and alignment

Structures may contain padding bytes so members align on sensible boundaries. As a result, the size of a struct can be larger than the sum of its members.

#include <stdio.h>

struct Mixed {
    char c;      // 1 byte
    int i;       // 4 bytes (3 padding bytes before it)
    char d;      // 1 byte (3 padding after, to align the struct)
};

int main(void) {
    printf("%zu\n", sizeof(struct Mixed));   // often 12
    return 0;
}
Output
12

Order members from largest to smallest to reduce padding. Use #pragma pack only when a binary format requires it.

Common mistakes

  • Forgetting the semicolon after the closing brace of a struct definition.
  • Using == to compare structures — C does not define it. Compare members or use memcmp (careful with padding).
  • Assigning to a string member with = — use strcpy.
  • Using . on a pointer or -> on a value.
  • Assuming sizeof(struct) equals the sum of member sizes.
