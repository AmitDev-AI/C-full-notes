C File Handling

Files let a program keep data after it exits. C uses a FILE * stream for both text and binary files, with functions from <stdio.h>.

The four steps

  1. Open the file with fopen.
  2. Check that the returned pointer is not NULL.
  3. Read or write using the stream functions.
  4. Close the file with fclose.

Warning: "w" erases the file immediately. Use "a" if you want to keep existing content.

Writing a text file

#include <stdio.h>

int main(void) {
    FILE *fp = fopen("notes.txt", "w");

    if (fp == NULL) {
        perror("Could not open notes.txt");
        return 1;
    }

    fprintf(fp, "C file handling\n");
    fprintf(fp, "Line %d\n", 2);
    fputs("Written with fputs\n", fp);

    fclose(fp);
    printf("File written.\n");
    return 0;
}
Output
File written.

Reading a text file

Line by line with fgets

#include <stdio.h>

int main(void) {
    FILE *fp = fopen("notes.txt", "r");
    char line[256];

    if (fp == NULL) {
        perror("Could not open notes.txt");
        return 1;
    }

    while (fgets(line, sizeof(line), fp) != NULL) {
        printf("%s", line);
    }

    fclose(fp);
    return 0;
}
Output
C file handling
Line 2
Written with fputs

Formatted reading with fscanf

#include <stdio.h>

int main(void) {
    FILE *fp = fopen("scores.txt", "r");
    int score;
    int count = 0;
    double sum = 0;

    if (fp == NULL) {
        perror("scores.txt");
        return 1;
    }

    while (fscanf(fp, "%d", &score) == 1) {
        sum += score;
        count++;
    }

    fclose(fp);

    if (count > 0) {
        printf("average = %.2f\n", sum / count);
    }
    return 0;
}

fscanf returns the number of items read, or EOF when input runs out. Always check it.

Appending to a file

FILE *fp = fopen("log.txt", "a");
if (fp != NULL) {
    fprintf(fp, "Program started\n");
    fclose(fp);
}

Binary files

Text functions stop at whitespace and translate newlines. For raw bytes — images, structs, arrays — use fread and fwrite.

#include <stdio.h>

typedef struct {
    int id;
    double price;
} Record;

int main(void) {
    Record out[2] = {{1, 9.99}, {2, 19.50}};
    Record in[2];

    FILE *fp = fopen("data.bin", "wb");
    if (fp == NULL) return 1;

    size_t written = fwrite(out, sizeof(Record), 2, fp);
    fclose(fp);

    if (written != 2) {
        printf("Write failed\n");
        return 1;
    }

    fp = fopen("data.bin", "rb");
    if (fp == NULL) return 1;

    size_t read = fread(in, sizeof(Record), 2, fp);
    fclose(fp);

    for (size_t i = 0; i < read; i++) {
        printf("%d: %.2f\n", in[i].id, in[i].price);
    }
    return 0;
}
Output
1: 9.99
2: 19.50

Binary files are not portable across systems with different type sizes or endianness. For data exchange, prefer a text format such as CSV or JSON.

Moving around: fseek and ftell

FunctionPurpose
fseek(fp, offset, origin)Move the position indicator
ftell(fp)Current position in bytes
rewind(fp)Go back to the start
feof(fp)Non-zero after an attempted read past the end
ferror(fp)Non-zero if an error occurred
clearerr(fp)Clear EOF and error flags

Origins: SEEK_SET (start), SEEK_CUR (current), SEEK_END (end).

#include <stdio.h>

int main(void) {
    FILE *fp = fopen("notes.txt", "r");
    if (fp == NULL) return 1;

    fseek(fp, 0, SEEK_END);
    long size = ftell(fp);
    fseek(fp, 0, SEEK_SET);

    printf("file size = %ld bytes\n", size);
    fclose(fp);
    return 0;
}
Output
file size = 48 bytes

Error handling

perror prints your message plus the system's reason, using errno.

#include <stdio.h>
#include <errno.h>
#include <string.h>

FILE *fp = fopen("missing.txt", "r");
if (fp == NULL) {
    perror("fopen");
    printf("errno = %d (%s)\n", errno, strerror(errno));
    return 1;
}
Output
fopen: No such file or directory
errno = 2 (No such file or directory)

Reading and writing at the same time

When a stream is opened for both reading and writing, you must call fflush or a positioning function between a write and a read (and vice versa). Forgetting this is undefined behaviour.

Closing files

fclose flushes buffered output and releases the stream. If you forget, data may be lost when the program exits abnormally, and you may run out of file handles.

Tip: always check fopen, fread, fwrite and fclose. Disk-full, permission and missing-file errors are normal, not exceptional.
